cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(VirtualMouseInput VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(VMI_ASAN "Instrument the application with AddressSanitizer" OFF)
if(VMI_ASAN)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /PDBALTPATH:%_PDB%)
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(MouseProtocol STATIC App/Mouse.cpp Vdevice/Vdevice.cpp)
target_include_directories(MouseProtocol PUBLIC App Vdevice)
add_executable(VirtualMouseInput WIN32 App/Main.cpp App/Session.cpp App/App.rc)
target_link_libraries(VirtualMouseInput PRIVATE MouseProtocol d2d1 dwrite windowscodecs dwmapi comctl32 shell32 ole32 advapi32)
target_link_options(VirtualMouseInput PRIVATE "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/App/App.manifest")
